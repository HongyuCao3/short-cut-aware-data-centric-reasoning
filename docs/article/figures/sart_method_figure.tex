% =====================================================================
% Panel 1: SCORE  —  boundary x-range (-2.55, 2.65), center 0.05
% =====================================================================
\node[title] (T1) at (0.05, 2.30) {Step 1 (Score):\\Gradient Signatures};
\draw[panelbox] (-2.55, 1.70) rectangle (2.65, -1.70);

% Input chip and label (top-left)
\node[font=\scriptsize, anchor=south] at (-2.10, 0.92) {sample $s$};
\node[inputchip] (gsIn) at (-2.10, 0.50) {$\mathbf{g}_s$};

% --- A(s) sub-icon (top, alignment / cosine) ---
\node[iconbox, minimum width=2.40cm, minimum height=1.05cm] (boxA) at (-0.20, 0.78) {};
\node[font=\footnotesize\bfseries, anchor=west, text=black!85] at (-1.35, 1.05) {$A(s)$};
\node[font=\tiny, anchor=west, text=black!55] at (-1.35, 0.83) {alignment};
% small angular vectors, centered in the box right of the header
\coordinate (origA) at (-0.20, 0.55);
\draw[gsarrow] (origA) -- ++(60:0.55);

\node[font=\tiny, anchor=south, text=orange!75!black] at ($(origA) + (60:0.55)$) {$\mathbf{g}_s$};
\node[font=\tiny, anchor=west, text=blue!55!black] at ($(origA) + (15:0.70)$) {$\mathbf{g}_{\mathcal V}$};

% --- R(s) sub-icon (bottom, answer-token concentration) ---
\node[iconbox, minimum width=2.40cm, minimum height=1.05cm] (boxR) at (-0.20, -0.42) {};
\node[font=\footnotesize\bfseries, anchor=west, text=black!85] at (-1.35, -0.20) {$R(s)$};
\node[font=\tiny, anchor=west, text=black!55] at (-1.35, -0.42) {answer mass};

\node[reasonbar] at ( 0.57, -0.45) {};
\node[ansbar, minimum width=0.32cm, minimum height=0.55cm] (ansbar) at (0.86, -0.45) {};
\node[font=\tiny, anchor=center, text=black!55] at (0.20, -0.85) {energy concentrates on answer};

% S(s) output chip on the right of the panel
\node[scorechip] (Sout) at (2.20, 0.12) {$S(s)$};

% Arrows: g_s flows into both icons
\draw[thinflow] (gsIn.east) -| ($(boxA.west)+(0.0,0)$);
\draw[thinflow] (gsIn.east) -| ($(boxR.west)+(0.0,0)$);
% Sub-icons combine into S(s)
\draw[flow] (boxA.east) -| (Sout.north);
\draw[flow] (boxR.east) -| (Sout.south);

% Hyperparameter footer
\node[hyptag] at (0.05, -1.50) {$\tau_A{=}0.3,\ \tau_R{=}0.5,\ \alpha{=}\beta{=}1.0$};

% =====================================================================
% Panel 2: CORRECT  —  boundary x-range (3.05, 9.35), center 6.20  (NOVEL)
% =====================================================================
\node[title] (T2) at (6.20, 2.30) {Step 2 (Correct):\\Magnitude $+$ Direction};
\draw[panelbox] (3.05, 1.70) rectangle (9.35, -1.70);
\node[novelflag] at (6.20, 1.35) {Novel: SART correction};

% --- 2a: magnitude reweight (left half) ---
\node[novelbox, minimum width=2.55cm, minimum height=2.00cm] (boxRW) at (4.55, -0.20) {};

% pre arrow: full-length g_s
\draw[gsarrow, line width=1.1pt] (3.55, 0.30) -- (5.05, 0.30) node[right=-1pt, font=\tiny] {$\mathbf{g}_s$};

\node[novelchip, font=\scriptsize] (wchip) at (4.55, -0.20) {$w(s){=}e^{-\lambda S}$};
% post arrow: shorter, signaling shrunk magnitude
\draw[gsarrow, line width=0.7pt] (3.85, -0.75) -- (4.65, -0.75) node[right=-1pt, font=\tiny] {$w\,\mathbf{g}_s$};
\node[font=\tiny, anchor=center, text=black!55] at (4.55, -1.05) {magnitude shrinks};

% --- 2b: conflict-only surgery (right half) ---
\node[novelbox, minimum width=3.10cm, minimum height=2.00cm] (boxSurg) at (7.65, -0.20) {};
\node[font=\scriptsize\bfseries, anchor=north] at (7.65, 0.78) {Conflict-only surgery};

\coordinate (surgOrig) at (6.65, -0.70);
% g_V points right-up (validation direction)
\draw[gvarrow] (surgOrig) -- ++(15:1.10) coordinate (gvTip);
\node[font=\tiny, anchor=south west, inner sep=1pt, text=blue!55!black]
     at ($(surgOrig)+(15:1.10)$) {$\mathbf{g}_{\mathcal V}$};
% g_s points up-left (anti-aligned)
\draw[gsarrow] (surgOrig) -- ++(125:0.95) coordinate (gsTip);
\node[font=\tiny, anchor=east, inner sep=1pt, text=orange!75!black]
     at ($(surgOrig)+(125:0.95)$) {$\mathbf{g}_s$};
% keep component along +g_V
\draw[surgerykeep] (surgOrig) -- ++(15:0.40)
     node[right=-2pt, font=\tiny, text=green!40!black] {keep};
% conflicting component (project out): from gsTip back along -g_V direction
\draw[surgerydrop] (gsTip) -- ($(surgOrig) + (15:0.40)$);
\node[font=\tiny, text=red!70!black, anchor=south] at (7.40, 0.20) {project out};
% gate condition (center-bottom of surgery sub-panel)
\node[font=\tiny, align=center, draw=red!50, fill=red!10, rounded corners=1pt,
      inner sep=2pt] (gate) at (8.55, -0.95)
     {gate fires iff\\$\mathbf{g}_s\!\cdot\!\mathbf{g}_{\mathcal V} \!<\! 0$};

% Hyperparameter footer
\node[hyptag] at (6.20, -1.50) {$\lambda{=}2.0,\ \ \gamma{=}1.0\ \text{(full projection)}$};

\node[novelchip, font=\footnotesize] (gprime) at (10.30, 0.50) {$w(s)\,\mathbf{g}'_s$};
